32inline std::string extract_simple_list_format(const JsonObjectMap& map) {
33 auto fIt = map.find("format");

44inline std::optional<std::size_t> extract_simple_list_length(const JsonObjectMap& map) {
45 std::optional<std::size_t> output;
46 auto lIt = map.find("length");
47 if (lIt != map.end()) {
48 const auto& val = lIt->second;
49 if (val->type() != millijson::NUMBER) {
50 throw std::runtime_error("expected a JSON number");
51 }
52 const auto num = reinterpret_cast<millijson::Number*>(val.get())->value();
53 if (num != std::trunc(num)) {
54 throw std::runtime_error("expected an integer");
55 }
56 output = sanisizer::from_float<std::size_t>(num);
57 }
58 return output;
