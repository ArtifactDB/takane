107inline void validate(const std::filesystem::path& path, const ObjectMetadata& metadata, const Options& options) {
108 auto cIt = options.custom_validate.find(metadata.type);
109
110 if (cIt != options.custom_validate.end()) {
111 try {
112 (cIt->second)(path, metadata, options);
113 } catch (...) {
114 std::throw_with_nested(std::runtime_error("failed to validate '" + metadata.type + "' object"));
115 }
116
117 } else {
118 static const auto validate_registry = default_validate_registry();
119 auto vrIt = validate_registry.find(metadata.type);
120 if (vrIt == validate_registry.end()) {
121 throw std::runtime_error("no registered validation function for object type '" + metadata.type + "'");
122 }
123
124 // Can't easily roll this out, as this is const and the above is not.
125 try {
126 (vrIt->second)(path, metadata, options);
127 } catch (...) {
128 std::throw_with_nested(std::runtime_error("failed to validate '" + metadata.type + "' object"));
129 }
130 }
131
132 if (options.custom_global_validate) {
133 try {
134 options.custom_global_validate(path, metadata, options);
135 } catch (...) {
136 std::throw_with_nested(std::runtime_error("failed additional validation for '" + metadata.type + "'"));
137 }
138 }
139}
