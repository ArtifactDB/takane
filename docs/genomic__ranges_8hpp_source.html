95inline bool start_plus_width_exceeds_int64(std::int64_t start, std::uint64_t width) {
96 constexpr std::uint64_t end_limit = std::numeric_limits<std::int64_t>::max();
97 if (start >= 0) {
98 // That is, does 'start + width > end_limit'?
99 // No underflow as 'start <= end_limit', so 'end_limit - start >= 0'.
100 return sanisizer::is_less_than(end_limit - start, width);
101 }
102
103 const std::uint64_t abs_start = static_cast<std::uint64_t>(-(1 + start)) + 1; // effectively '-start' but avoid overflow of the signed type.
104 if (abs_start >= width) {
105 // 'start + width == width - abs_start', which would be non-positive in this case.
106 // So it can't possibly exceed end_limit.
107 return false;
108 }
109
110 return sanisizer::is_less_than(end_limit, width - abs_start);
111}

154
155 H5::DataSet id_handle;

166 id_space.getSimpleExtentDims(&num_ranges);
167 } catch (...) {
168 std::throw_with_nested(std::runtime_error("failed to validate 'sequence'"));

172 try {

218 const auto id = id_stream.next();
219 if (sanisizer::is_greater_than_or_equal(id, num_sequences)) {
220 throw std::runtime_error("'sequence' must be less than the number of sequences in 'sequence_information'");
221 }
222
223 const auto start = start_stream.next();
224 const auto width = width_stream.next();
225
226 // If it's definitely non-circular, the start position should be positive.
227 const auto& circular = limits->circular[id];
228 if (circular.has_value() && !(*circular)) {
229 if (start < 1) {
230 throw std::runtime_error("non-positive 'start' position for a non-circular sequence");
231 }
232
233 const auto& length = limits->length[id];
234 if (length.has_value()) {

256 }

259 if (strand_space.getSimpleExtentNdims() != 1) {
260 throw std::runtime_error("expected a 1-dimensional dataset");
261 }
262 hsize_t nstrand;

268 ritsuko::hdf5::Stream1dNumericDataset<std::int32_t> strand_stream(&strand_handle, num_ranges, opt);

271 [&](hsize_t, std::int32_t x) -> void {
272 if (x < -1 || x > 1) {
273 throw std::runtime_error("entries should be one of 0, -1, or 1");
274 }
275 }
276 );
277 } catch (...) {
278 std::throw_with_nested(std::runtime_error("failed to validate 'strand'"));
279 }
280
281 validate_names(ghandle, "name", num_ranges, options.hdf5_buffer_size);
282 } catch (...) {
283 std::throw_with_nested(std::runtime_error("failed to validate '" + type_name + "' in 'ranges.h5'"));
284 }

286 validate_mcols(path, "range_annotations", num_ranges, options);
287 validate_metadata(path, "other_annotations", options);
288}
