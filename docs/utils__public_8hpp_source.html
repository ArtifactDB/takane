48inline ObjectMetadata reformat_object_metadata(millijson::Base* raw) {
49 if (raw->type() != millijson::OBJECT) {
50 throw std::runtime_error("metadata should be a JSON object");
51 }
52
53 ObjectMetadata output;
54 output.other = std::move(reinterpret_cast<millijson::Object*>(raw)->value());
55
56 auto tIt = output.other.find("type");
57 if (tIt == output.other.end()) {
58 throw std::runtime_error("metadata should have a 'type' property");
59 }
60
61 const auto& tval = tIt->second;
62 if (tval->type() != millijson::STRING) {
63 throw std::runtime_error("metadata should have a 'type' string");
64 }
65
66 output.type = std::move(reinterpret_cast<millijson::String*>(tval.get())->value());
67 output.other.erase(tIt);
68 return output;
69}
