37inline void validate_spatial_coordinates(const std::filesystem::path& coord_path, std::size_t ncols, const Options& options) {
38 auto coord_meta = read_object_metadata(coord_path);
39 if (!derived_from(coord_meta.type, "dense_array", options)) {
40 throw std::runtime_error("expected a dense array");
41 }
42
43 // Validating the coordinates; currently these must be a dense array of
44 // points, but could also be polygons/hulls in the future.
45 validate(coord_path, coord_meta, options);
46
47 auto cdims = dimensions(coord_path, coord_meta, options);
48 if (cdims.size() != 2) {
49 throw std::runtime_error("expected a 2-dimensional dense array");
50 } else if (cdims[1] != 2 && cdims[1] != 3) {
51 throw std::runtime_error("expected 2 or 3 columns");
52 } else if (cdims[0] != ncols) {
53 throw std::runtime_error("number of rows in should equal the number of columns in the 'spatial_experiment'");
54 }
55
56 // Checking that the values are numeric.
57 H5::H5File handle(coord_path / "array.h5", H5F_ACC_RDONLY);
58 auto ghandle = handle.openGroup("dense_array");
59 auto dhandle = ghandle.openDataSet("data");
60 auto dclass = dhandle.getTypeClass();
61 if (dclass != H5T_INTEGER && dclass != H5T_FLOAT) {
62 throw std::runtime_error("values should be numeric");
63 }
64}
